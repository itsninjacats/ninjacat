// The requests behind api/tests/Intake.Tests/Fixtures/trace: payloads the
// Go tests of router_trace.go did not send through HTTP (the idx wire format,
// real sketches, every msgp error, unknown Data Streams fields), recorded by
// fixtures_test.go's recorder the same way the fixtures in ../go were. To
// record again, put this file in server/intake as a _test.go file.

func portSketch(t *testing.T, m mapping.IndexMapping, values ...float64) []byte {
	t.Helper()
	sk := ddsketch.NewDDSketch(m, store.NewDenseStore(), store.NewDenseStore())
	for _, v := range values {
		if err := sk.Add(v); err != nil {
			t.Fatalf("add %g: %v", v, err)
		}
	}
	raw, err := proto.Marshal(sk.ToProto())
	if err != nil {
		t.Fatal(err)
	}
	return raw
}

func portProto(t *testing.T, m proto.Message) []byte {
	t.Helper()
	raw, err := proto.Marshal(m)
	if err != nil {
		t.Fatal(err)
	}
	return raw
}

func TestPortStats(t *testing.T) {
	a, _ := newTestServer(t)
	e := newTestEngine(t, a, a.routeTrace)

	logMap, _ := mapping.NewLogarithmicMapping(0.01)
	linMap, _ := mapping.NewLinearlyInterpolatedMapping(0.01)
	cubMap, _ := mapping.NewCubicallyInterpolatedMapping(0.01)
	logOff, _ := mapping.NewLogarithmicMappingWithGamma(1.05, 3.5)

	many := make([]float64, 0, 60)
	for i := 0; i < 60; i++ {
		many = append(many, 1000*math.Pow(1.37, float64(i%24))+float64(i))
	}

	sparse := portProto(t, &sketchpb.DDSketch{
		Mapping:        &sketchpb.IndexMapping{Gamma: 1.02020202020202, IndexOffset: 0},
		PositiveValues: &sketchpb.Store{BinCounts: map[int32]float64{-5: 2, 1000: 1.5, 3: 0, 7: 4}, ContiguousBinCounts: []float64{1, 0, 2}, ContiguousBinIndexOffset: 6},
		NegativeValues: &sketchpb.Store{BinCounts: map[int32]float64{12: 3}},
		ZeroCount:      2,
	})
	zeroOnly := portProto(t, &sketchpb.DDSketch{Mapping: &sketchpb.IndexMapping{Gamma: 1.02}, ZeroCount: 4})
	negOnly := portProto(t, &sketchpb.DDSketch{
		Mapping:        &sketchpb.IndexMapping{Gamma: 1.02, Interpolation: sketchpb.IndexMapping_CUBIC},
		NegativeValues: &sketchpb.Store{ContiguousBinCounts: []float64{1, 2, 3}, ContiguousBinIndexOffset: -4},
	})
	noMapping := portProto(t, &sketchpb.DDSketch{ZeroCount: 4})
	badGamma := portProto(t, &sketchpb.DDSketch{Mapping: &sketchpb.IndexMapping{Gamma: 1}, ZeroCount: 4})
	quadratic := portProto(t, &sketchpb.DDSketch{Mapping: &sketchpb.IndexMapping{Gamma: 1.5, Interpolation: sketchpb.IndexMapping_QUADRATIC}, ZeroCount: 4})
	emptyWithMapping := portProto(t, &sketchpb.DDSketch{Mapping: &sketchpb.IndexMapping{Gamma: 1.5}})

	group := func(name string, ok, errS []byte) *pb.ClientGroupedStats {
		return &pb.ClientGroupedStats{Service: "checkout", Name: name, OkSummary: ok, ErrorSummary: errS}
	}

	payload := &pb.StatsPayload{
		AgentHostname: "agent-1", AgentEnv: "prod", AgentVersion: "7.58.2", ClientComputed: true, SplitPayload: true,
		Stats: []*pb.ClientStatsPayload{
			{
				Hostname: "pod-1", Env: "prod", Version: "1.4.0", Lang: "python", TracerVersion: "2.9.0",
				RuntimeID: "runtime-1", Sequence: 7, AgentAggregation: "distributions", Service: "checkout", ContainerID: "abc123",
				Tags:         []string{"kube_service:a", "kube_service:b", "bare", "url:http://x:1"},
				GitCommitSha: "deadbeef", ImageTag: "v1.4.0", ProcessTags: "entrypoint:gunicorn", ProcessTagsHash: 18446744073709551615,
				Stats: []*pb.ClientStatsBucket{
					{
						Start: 1_700_000_000_000_000_001, Duration: 10_000_000_000, AgentTimeShift: -500,
						Stats: []*pb.ClientGroupedStats{
							{
								Service: "checkout", Name: "flask.request", Resource: "GET /pay",
								HTTPStatusCode: 200, Type: "web", DBType: "pg", Hits: 120, Errors: 3,
								Duration: 900_000_000, Synthetics: true, TopLevelHits: 119, SpanKind: "server",
								IsTraceRoot: pb.Trilean_TRUE, GRPCStatusCode: "0", HTTPMethod: "GET", HTTPEndpoint: "/pay",
								ServiceSource: "opt", PeerTags: []string{"db.hostname:pg-1"},
								SpanDerivedPrimaryTags: []string{"dc:eu"}, AdditionalMetricTags: []string{"region:eu"},
								OkSummary:    portSketch(t, logMap, 1e6, 2e6, 3e6),
								ErrorSummary: []byte("not a sketch"),
							},
							{Service: "checkout", Name: "false-root", IsTraceRoot: pb.Trilean_FALSE},

			{Service: "second", Stats: []*pb.ClientStatsBucket{{Stats: []*pb.ClientGroupedStats{{Name: "x", Hits: 1}}}}},

	}
	body, err := payload.MarshalMsg(nil)
	if err != nil {
		t.Fatal(err)
	}
	post(t, e, "/api/v0.2/stats", body) // 001

	// Hand-built bodies.
	type w = []byte
	hdr := func(b w, n uint32) w { return msgp.AppendMapHeader(b, n) }
	str := func(b w, s string) w { return msgp.AppendString(b, s) }
	arr := func(b w, n uint32) w { return msgp.AppendArrayHeader(b, n) }

	// one group nested three levels down, with the given tail
	nested := func(tail func(b w) w) w {
		b := hdr(nil, 2)
		b = str(b, "AgentHostname")
		b = str(b, "h")
		b = str(b, "Stats")
		b = arr(b, 2)
		b = hdr(b, 1) // client 0
		b = str(b, "Service")
		b = str(b, "c0")
		b = hdr(b, 1) // client 1
		b = str(b, "Stats")
		b = arr(b, 1)
		b = hdr(b, 1) // bucket 0
		b = str(b, "Stats")
		b = arr(b, 3)
		b = hdr(b, 1) // group 0
		b = str(b, "Name")
		b = str(b, "g0")
		b = msgp.AppendNil(b) // group 1 nil
		return tail(b)        // group 2
	}

	cases := []w{
		// 002 wrong type deep
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "Hits"); return str(b, "x") }),
		// 003 negative uint
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "Hits"); return msgp.AppendInt64(b, -5) }),
		// 004 uint32 overflow
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "HTTPStatusCode"); return msgp.AppendUint64(b, 1<<40) }),
		// 005 int32 overflow
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "IsTraceRoot"); return msgp.AppendInt64(b, 1<<40) }),
		// 006 truncated
		nested(func(b w) w { b = hdr(b, 2); b = str(b, "Hits"); return msgp.AppendUint64(b, 5) }),
		// 007 unknown key skipped, nested value
		nested(func(b w) w {
			b = hdr(b, 2)
			b = str(b, "Brand")
			b = hdr(b, 1)
			b = str(b, "k")
			b = arr(b, 2)
			b = msgp.AppendFloat64(b, 1.5)
			b = msgp.AppendBytes(b, []byte{1, 2})
			b = str(b, "Hits")
			return msgp.AppendInt8(b, 9)
		}),
		// 008 unknown key with an invalid byte inside
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "Brand"); b = arr(b, 1); return append(b, 0xc1) }),
		// 009 bin as map key, str for OkSummary
		nested(func(b w) w {
			b = hdr(b, 2)
			b = msgp.AppendBytes(b, []byte("Name"))
			b = str(b, "binkey")
			b = str(b, "OkSummary")
			return str(b, "text")
		}),
		// 010 top-level nil
		msgp.AppendNil(nil),
		// 011 trailing bytes after the payload
		append(nested(func(b w) w { b = hdr(b, 1); b = str(b, "Hits"); return msgp.AppendUint64(b, 5) }), 0xc1, 0xc1),
		// 012 uint64 above MaxInt64 for a signed field
		func() w {
			b := hdr(nil, 1)
			b = str(b, "Stats")
			b = arr(b, 1)
			b = hdr(b, 1)
			b = str(b, "Stats")
			b = arr(b, 1)
			b = hdr(b, 1)
			b = str(b, "AgentTimeShift")
			return msgp.AppendUint64(b, 1<<63)
		}(),
		// 013 float for Hits
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "Hits"); return msgp.AppendFloat64(b, 2) }),
		// 014 a negative fixint for a uint
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "Errors"); return append(b, 0xff) }),
		// 015 bool given a number
		func() w { b := hdr(nil, 1); b = str(b, "ClientComputed"); return msgp.AppendInt(b, 1) }(),
		// 016 map header instead of string array
		nested(func(b w) w {
			b = hdr(b, 1)
			b = str(b, "PeerTags")
			b = arr(b, 2)
			b = str(b, "a")
			return msgp.AppendInt(b, 3)
		}),
		// 017 integer key
		func() w { b := hdr(nil, 1); return msgp.AppendInt(b, 1) }(),
		// 018 a float32 where a string belongs, three levels down
		nested(func(b w) w { b = hdr(b, 1); b = str(b, "Service"); return msgp.AppendFloat32(b, 1.5) }),
		// 019 the invalid byte as a string
		func() w { b := hdr(nil, 1); b = str(b, "AgentEnv"); return append(b, 0xc1) }(),
		// 020 the invalid byte as a top-level value
		{0xc1},
		// 021 ext where an array belongs
		func() w { b := hdr(nil, 1); b = str(b, "Stats"); return append(b, 0xd4, 0x05, 0x00) }(),
		// 022 str32 with a forged length
		func() w {
			b := hdr(nil, 1)
			b = str(b, "AgentEnv")
			return append(b, 0xdb, 0xff, 0xff, 0xff, 0xff, 'a')
		}(),
		// 023 array16 claiming more than is there
		func() w { b := hdr(nil, 1); b = str(b, "Stats"); return append(b, 0xdc, 0x00, 0x05) }(),
		// 024 unknown key nested 200 deep, skipped
		func() w {
			b := hdr(nil, 2)
			b = str(b, "X")
			for i := 0; i < 200; i++ {
				b = arr(b, 1)
			}
			b = msgp.AppendInt(b, 1)
			b = str(b, "AgentEnv")
			return str(b, "prod")
		}(),
	}
	for _, c := range cases {
		post(t, e, "/api/v0.2/stats", c)
	}
}

func TestPortPipelineStats(t *testing.T) {
	a, _ := newTestServer(t)
	e := newTestEngine(t, a, a.routeTrace)

	send := func(body []byte) {
		req := httptest.NewRequest(http.MethodPost, "/api/v0.1/pipeline_stats?x=1&x=2", bytes.NewReader(body))
		req.Header.Set("Dd-Api-Key", testAPIKey)
		req.Header.Set("Content-Type", "application/msgpack")
		req.Header.Set("Via", "trace-agent 7.58.2")
		req.Header.Set("X-Datadog-Additional-Tags", "_dd.tags.container:cid")
		w := httptest.NewRecorder()
		e.ServeHTTP(w, req)
	}

	send(portDSMFull(t)) // 001

	type w = []byte
	hdr := func(b w, n uint32) w { return msgp.AppendMapHeader(b, n) }
	str := func(b w, s string) w { return msgp.AppendString(b, s) }
	arr := func(b w, n uint32) w { return msgp.AppendArrayHeader(b, n) }

	point := func(tail func(b w) w) w {
		b := hdr(nil, 1)
		b = str(b, "Stats")
		b = arr(b, 1)
		b = hdr(b, 1)
		b = str(b, "Stats")
		b = arr(b, 1)
		return tail(b)
	}

	cases := []w{
		// 002 nil body value
		msgp.AppendNil(nil),
		// 003 wrong type at payload level
		func() w { b := hdr(nil, 1); b = str(b, "Env"); return msgp.AppendInt(b, 5) }(),
		// 004 wrong type in a point
		point(func(b w) w { b = hdr(b, 1); b = str(b, "Hash"); return str(b, "x") }),
		// 005 negative hash
		point(func(b w) w { b = hdr(b, 1); b = str(b, "Hash"); return msgp.AppendInt64(b, -3) }),
		// 006 truncated inside a point
		point(func(b w) w { b = hdr(b, 2); b = str(b, "Hash"); return msgp.AppendUint64(b, 3) }),
		// 007 truncated string
		func() w { b := hdr(nil, 1); b = str(b, "Env"); return append(b, 0xa5, 'a') }(),
		// 008 unknown with invalid byte nested
		point(func(b w) w { b = hdr(b, 1); b = str(b, "New\"x"); b = arr(b, 1); return append(b, 0xc1) }),
		// 009 unknown map with a bin key
		func() w {
			b := hdr(nil, 1)
			b = str(b, "New")
			b = hdr(b, 1)
			b = msgp.AppendBytes(b, []byte("k"))
			return msgp.AppendInt(b, 1)
		}(),
		// 010 NaN in an unknown field
		point(func(b w) w {
			b = hdr(b, 2)
			b = str(b, "Hash")
			b = msgp.AppendUint64(b, 3)
			b = str(b, "New")
			return msgp.AppendFloat64(b, math.NaN())
		}),
		// 011 str where bytes belong
		point(func(b w) w { b = hdr(b, 1); b = str(b, "PathwayLatency"); return str(b, "x") }),
		// 012 backlog value too large
		func() w {
			b := hdr(nil, 1)
			b = str(b, "Stats")
			b = arr(b, 1)
			b = hdr(b, 1)
			b = str(b, "Backlogs")
			b = arr(b, 1)
			b = hdr(b, 1)
			b = str(b, "Value")
			return msgp.AppendUint64(b, 1<<63)
		}(),
		// 013 array claiming more than is there
		func() w { b := hdr(nil, 1); b = str(b, "ProcessTags"); b = arr(b, 3); return str(b, "one") }(),
		// 014 bin key at the top level
		func() w { b := hdr(nil, 1); b = msgp.AppendBytes(b, []byte("Env")); return str(b, "x") }(),
		// 015 unknown array longer than the body
		func() w { b := hdr(nil, 1); b = str(b, "New"); return append(b, 0xdc, 0xff, 0xff) }(),
		// 016 float32 NaN and nested map in unknown, with a duplicate key
		func() w {
			b := hdr(nil, 2)
			b = str(b, "Stats")
			b = arr(b, 1)
			b = hdr(b, 1)
			b = str(b, "Stats")
			b = arr(b, 1)
			b = hdr(b, 1)
			b = str(b, "Hash")
			b = msgp.AppendUint64(b, 3)
			b = str(b, "New")
			b = hdr(b, 2)
			b = str(b, "k")
			b = msgp.AppendInt(b, 1)
			b = str(b, "k")
			return msgp.AppendInt(b, 2)
		}(),
		// 017 a one-byte body that is a bin lead
		{0xc4},
		// 018 bytes field with one byte left
		point(func(b w) w { b = hdr(b, 1); b = str(b, "EdgeLatency"); return append(b, 0xc4) }),
		// 019 bytes field truncated data
		point(func(b w) w { b = hdr(b, 1); b = str(b, "EdgeLatency"); return append(b, 0xc4, 0x05, 0x01) }),
		// 020 map16 header truncated
		{0xde, 0x00},
		// 021 trailing bytes after the payload
		append(point(func(b w) w { b = hdr(b, 1); b = str(b, "Hash"); return msgp.AppendUint64(b, 3) }), 0xc1),
		// 022 str32 with a forged length
		func() w { b := hdr(nil, 1); b = str(b, "Env"); return append(b, 0xdb, 0xff, 0xff, 0xff, 0xff, 'a') }(),
	}
	for _, c := range cases {
		send(c)
	}
}

func nestedJSON(n int) string {
	var b bytes.Buffer
	for i := 0; i < n; i++ {
		b.WriteString("[")
	}
	for i := 0; i < n; i++ {
		b.WriteString("]")
	}
	return b.String()
}
